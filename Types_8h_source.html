60// Enum class doesn't support bitwise ops natively. These allow bit-testing
61// e.g. (transformClass & TransformClass::POINT) != TransformClass::NONE
62inline TransformClass operator|(TransformClass a, TransformClass b)
63{
64 return static_cast<TransformClass>(static_cast<uint8_t>(a) | static_cast<uint8_t>(b));
65}
66inline TransformClass operator&(TransformClass a, TransformClass b)
67{
68 return static_cast<TransformClass>(static_cast<uint8_t>(a) & static_cast<uint8_t>(b));
69}
70// e.g. hasFlag(prim.transformType(), TransformClass::POINT)
71inline bool hasFlag(TransformClass value, TransformClass flag)
72{
73 return (value & flag) != TransformClass::NONE;
74}
