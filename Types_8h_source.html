62// Enum class doesn't support bitwise ops natively. These allow bit-testing
63// e.g. (transformClass & TransformClass::POINT) != TransformClass::NONE
64inline TransformClass operator|(TransformClass a, TransformClass b)
65{
66 return static_cast<TransformClass>(static_cast<uint8_t>(a) | static_cast<uint8_t>(b));
67}
68inline TransformClass operator&(TransformClass a, TransformClass b)
69{
70 return static_cast<TransformClass>(static_cast<uint8_t>(a) & static_cast<uint8_t>(b));
71}
72// e.g. hasFlag(prim.transformType(), TransformClass::POINT)
73inline bool hasFlag(TransformClass value, TransformClass flag)

190inline ValueType toValueType(Type type)
191{
192 switch (type)
193 {
194 case Type::FLOAT:
195 case Type::XYZ:
196 return ValueType::Float;
197 case Type::INT:
198 case Type::BOOL:
199 case Type::TOGGLE:
200 // Multiparm parameters (like ramp) use integers to represent their instance
201 // count and store the actual data in their instances.
202 case Type::RAMP:
203 // Spacers are purely visual and store an int nobody reads.
204 case Type::SPACER:
205 return ValueType::Int;
206 case Type::STRING:
207 case Type::DROPDOWN:
208 return ValueType::String;
209 case Type::GROUP:
210 return ValueType::Float;
211 }
212 return ValueType::Float;
213}
